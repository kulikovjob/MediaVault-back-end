<script>
  const fileTypes = {
    "Image": 1,
    "Video": 2,
    "Audio": 3,
    "Document": 4
  };

  fetch('/api/v1/media/files/')
    .then(response => {
      if (!response.ok) {
        throw new Error('Failed to fetch files');
      }
      return response.json();
    })
    .then(data => {
      const fileList = document.getElementById('file-list');
      if (data.status === 'success') {
        data.data.media.forEach(file => {
          const col = document.createElement('div');
          col.className = 'col-md-4';
          col.innerHTML = `
              <div class="card mb-4 shadow-sm">
                <div class="card-body">
                  <h5 class="card-title">${file.file_name}</h5>
                  <p class="card-text">ID: ${file.file_id}</p>
                  <p class="card-text">Upload Date: ${new Date(file.upload_date).toLocaleDateString()}</p>
                  <p class="card-text">Author: ${file.author_first_name} ${file.author_second_name}</p>
                  <p class="card-text">File Type: ${file.file_type}</p>
                  <p class="card-text">Rating: ${file.file_rating}</p>
                </div>
              </div>
            `;
          fileList.appendChild(col);
        });
      } else {
        throw new Error(data.message);
      }
    })
    .catch(error => {
      console.error('Error:', error);
      redirectToErrorPage(error.message);
    });

  function redirectToErrorPage(errorMessage) {
    const errorLink = document.createElement('a');
    errorLink.href = `/error.html?message=${errorMessage}`;
    errorLink.className = 'btn btn-danger mt-4';
    errorLink.textContent = 'View Error Details';
    document.body.appendChild(errorLink);
    setTimeout(() => {
      window.location.href = errorLink.href;
    }, 3000); // Redirect after 3 seconds
  }
</script>
